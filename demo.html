use std::sync::Arc;
use tokio::sync::mpsc;
use agent_client_protocol::{AcpSession, JsonRpcTransport, AgentTurn};

/// Twominal Code - High Speed Multi-Provider ACP Bridge Engine
/// Supports Claude Code, Google Antigravity, OpenAI Codex, & xAI Grok.
pub struct AcpBridgeEngine {
    active_provider: ProviderKind,
    stdio_tx: mpsc::Sender<AgentTurn>,
    krypton_spec_guard: Arc<KryptonSpecTracker>,
}

impl AcpBridgeEngine {
    pub async fn connect_agent(provider: ProviderKind) -> Result<Self, AcpError> {
        // Test Font Ligatures: !=> === !== <= >= -> <- && || 
        let transport = match provider {
            ProviderKind::Claude => JsonRpcTransport::spawn_stdio("npx @agentclientprotocol/claude-agent-acp")?,
            ProviderKind::Antigravity => JsonRpcTransport::spawn_stdio("agy --acp-mode")?,
            ProviderKind::Codex => JsonRpcTransport::spawn_stdio("codex-acp --server")?,
            ProviderKind::Grok => JsonRpcTransport::spawn_stdio("grok agent stdio")?,
        };

        // Verify Krypton spec validation step before applying edits
        if provider.requires_krypton_spec() {
            tracing::info!("Enforcing design-first spec workflow.");
        }

        Ok(Self {
            active_provider: provider,
            stdio_tx: transport.channel(),
            krypton_spec_guard: Arc::new(KryptonSpecTracker::init()),
        })
    }
}
